> **再帰的CTEがパブリックプレビューに**
> 再帰的共通テーブル式が全てのコンピュートタイプとSQLウェアハウスのプレビューチャンネルで利用で桐生用になりました。RECURSIVEとしてCTEを作成することで、自己参照してイテレーティブなSQLクエリーを可能にします。

以前はDatabricksランタイム17.0でしかサポートされていませんでしたが、今回のアップデートですべてのコンピュートタイプで再帰的CTE(Common Table Expression)がサポートされました。

https://qiita.com/taka_yayoi/items/c75be35e17861c926d0c

マニュアルはこちらです。

https://docs.databricks.com/aws/ja/sql/language-manual/sql-ref-syntax-qry-select-cte

プレビューチャンネルのSQLウェアハウスを使います。

![Screenshot 2025-07-07 at 9.20.00.png](https://qiita-image-store.s3.ap-northeast-1.amazonaws.com/0/1168882/d0f19a41-64da-4b81-a096-e37f5f2bb25c.png)

以前はなかったサンプルクエリーを動かします。

次の例は、有向グラフを走査し、その中に無限ループ（巡回）の存在をチェックする方法を示しています。

```sql
-- グラフテーブルが存在しない場合は作成
CREATE TABLE IF NOT EXISTS graph( f int, t int, label string );
-- グラフテーブルに値を挿入
INSERT INTO graph VALUES
    (1, 2, 'arc 1 -> 2'),
    (1, 3, 'arc 1 -> 3'),
    (2, 3, 'arc 2 -> 3'),
    (1, 4, 'arc 1 -> 4'),
    (4, 5, 'arc 4 -> 5'),
    (5, 1, 'arc 5 -> 1');
```

以下のようなテーブルが作成されます。

|f|t|label|
|---|---|---|
|1|2|arc 1 -> 2|
|1|3|arc 1 -> 3|
|2|3|arc 2 -> 3|
|1|4|arc 1 -> 4|
|4|5|arc 4 -> 5|
|5|1|arc 5 -> 1|


```sql
-- グラフを検索するための再帰CTE
WITH RECURSIVE search_graph(f, t, label, path, cycle) AS (
    -- 基本ケース: グラフテーブルからすべての行を選択し、pathとcycleを初期化
    SELECT *, array(struct(g.f, g.t)), false FROM graph g
    UNION ALL
    -- 再帰ケース: pathを拡張し、cycleをチェック
    SELECT g.*, path || array(struct(g.f, g.t)), array_contains(path, struct(g.f, g.t))
      FROM graph g, search_graph sg
      WHERE g.f = sg.t AND NOT cycle
)
SELECT path FROM search_graph;
```

結果として得られるのは、ループを回避しながら徐々に長くなる経路の集合であり、最終的には無限ループに陥らない経路となります。

すべてのノードを訪問し、シンク「3」で明確に停止する経路になります。深さ優先探索を行うには、最後の行を path カラムで並び替えるように変更します。

```sql
-- グラフテーブルが存在しない場合は作成
CREATE TABLE IF NOT EXISTS graph( f int, t int, label string );

-- グラフテーブルに値を挿入
INSERT INTO graph VALUES
    (1, 2, 'arc 1 -> 2'),
    (1, 3, 'arc 1 -> 3'),
    (2, 3, 'arc 2 -> 3'),
    (1, 4, 'arc 1 -> 4'),
    (4, 5, 'arc 4 -> 5'),
    (5, 1, 'arc 5 -> 1');

-- グラフを検索するための再帰CTE
WITH RECURSIVE search_graph(f, t, label, path, cycle) AS (
    -- 基本ケース: グラフテーブルからすべての行を選択し、pathとcycleを初期化
    SELECT *, array(struct(g.f, g.t)), false FROM graph g
    UNION ALL
    -- 再帰ケース: pathを拡張し、cycleをチェック
    SELECT g.*, path || array(struct(g.f, g.t)), array_contains(path, struct(g.f, g.t))
      FROM graph g, search_graph sg
      WHERE g.f = sg.t AND NOT cycle
)
SELECT * FROM search_graph ORDER BY path;
```

以下の例は、基本的な再帰的手法を示しています。

```sql
-- ループのような単純な例です。
WITH RECURSIVE r(col) AS (
    SELECT 'a'
    UNION ALL
    SELECT col || char(ascii(substr(col, -1)) + 1) FROM r WHERE LENGTH(col) < 10
  )
  SELECT * FROM r;
```

```sql
-- 別のループのような例です。
> WITH RECURSIVE t(n) AS (
    SELECT 'foo'
    UNION ALL
    SELECT n || ' bar' FROM t WHERE length(n) < 20
  )
  SELECT n AS is_text FROM t;
```

```sql
-- 組織ツリーの例
-- 'A'配下の全ての部門を抽出
CREATE TABLE department (
    id INTEGER, -- 部門ID
    parent_department INTEGER, -- 上位部門ID
    name string -- 部門名
  );

INSERT INTO department VALUES
    (0, NULL, 'ROOT'),
    (1, 0, 'A'),
    (2, 1, 'B'),
    (3, 2, 'C'),
    (4, 2, 'D'),
    (5, 0, 'E'),
    (6, 4, 'F'),
    (7, 5, 'G');

WITH RECURSIVE subdepartment AS (
    SELECT name as root_name, * FROM department WHERE name = 'A'
    UNION ALL
    SELECT sd.root_name, d.* FROM department AS d, subdepartment AS sd
     WHERE d.parent_department = sd.id
  )
  SELECT * FROM subdepartment ORDER BY name;
```

```sql
-- 二分木の例
-- もう一つの木の例として、2階層目のノード「2」と「3」から他のすべてのノードへのパスを求める。
CREATE TABLE tree(id INTEGER, parent_id INTEGER);
INSERT INTO tree
    VALUES (1, NULL), (2, 1), (3, 1), (4, 2), (5, 2), ( 6, 2), ( 7, 3), ( 8, 3),
           (9,    4), (10,4), (11,7), (12,7), (13,7), (14, 9), (15,11), (16,11);
```

```sql
WITH RECURSIVE t(id, path) AS (
    VALUES(1,cast(array() as array<Int>))
    UNION ALL
    SELECT tree.id, t.path || array(tree.id)
      FROM tree JOIN t ON (tree.parent_id = t.id)
  )
  SELECT t1.*, t2.*
    FROM t AS t1 JOIN t AS t2
      ON (t1.path[0] = t2.path[0] AND
          size(t1.path) = 1 AND
          size(t2.path) > 1)
  ORDER BY t1.id, t2.id;
```

```sql
-- 数独ソルバー (https://sqlite.org)
-- ハイライトされたテキストは、9x9の数独グリッドを行ごとにスキャンしたもので、空白が未入力値を表します。
> WITH RECURSIVE generate_series(gs) AS (
    SELECT 1
    UNION ALL
    SELECT gs+1 FROM generate_series WHERE gs < 9
  ),
  x( s, ind ) AS
    ( SELECT sud, position( ' ' IN sud )
      FROM  (SELECT '4  8725  5  64 213 29   8      6 73 1 8 2 4  97  15 2 3  2  1    659   28 2 37946'::STRING
        AS sud) xx
    UNION ALL
    SELECT substr( s, 1, ind - 1 ) || z || substr( s, ind + 1 ),
           position(' ' in repeat('x',ind) || substr( s, ind + 1 ) )
    FROM x,
     (SELECT gs::STRING AS z FROM generate_series) z
    WHERE ind > 0
      AND NOT EXISTS ( SELECT *
        FROM generate_series
        WHERE z.z = substr( s, ( (ind - 1 ) DIV 9 ) * 9 + gs, 1 )
           OR    z.z = substr( s, mod( ind - 1, 9 ) - 8 + gs * 9, 1 )
           OR    z.z = substr( s, mod( ( ( ind - 1 ) DIV 3 ), 3 ) * 3
                       + ( ( ind - 1 ) DIV 27 ) * 27 + gs
                       + ( ( gs - 1 ) DIV 3 ) * 6, 1 )
     )
  )
  SELECT s FROM x WHERE ind = 0;
```
